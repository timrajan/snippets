function pickRandom<T>(list: readonly T[]): T {
  return list[Math.floor(Math.random() * list.length)];
}

/** Returns a random first name, e.g. 'James'. */
export function getFirstName(): string {
  return pickRandom(FIRST_NAMES);
}

/** Returns a random last name, e.g. 'Hargreaves'. */
export function getLastName(): string {
  return pickRandom(LAST_NAMES);
}

/** Returns a random date of birth as a string, e.g. '21/12/1975'. */
export function getDOB(): string {
  return pickRandom(DOBS);
}

/** Convenience: a full random person. */
export function getRandomPerson(): { firstName: string; lastName: string; dob: string } {
  return { firstName: getFirstName(), lastName: getLastName(), dob: getDOB() };
}
