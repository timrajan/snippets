import { parseArgs } from 'node:util';

const ALLOWED_ENVS = ['DEV', 'SIT', 'UAT', 'PROD'] as const;
type Env = (typeof ALLOWED_ENVS)[number];

const { values } = parseArgs({
  options: {
    env: { type: 'string', short: 'e' },
  },
  strict: true, // errors on unknown flags like --evn
});

const env = (values.env ?? 'DEV').toUpperCase() as Env;

if (!ALLOWED_ENVS.includes(env)) {
  console.error(`Invalid --env "${values.env}". Use one of: ${ALLOWED_ENVS.join(', ')}`);
  process.exit(1);
}

console.log(`Seeding test data for ${env}...`);

import { parseArgs } from 'node:util';

export const ALLOWED_ENVS = ['DEV', 'SIT', 'UAT', 'PROD'] as const;
export type Env = (typeof ALLOWED_ENVS)[number];

function resolveEnv(): Env {
  const { values } = parseArgs({
    options: { env: { type: 'string', short: 'e' } },
    strict: false, // tolerate other flags when imported by different scripts
  });

  const raw = String(values.env ?? process.env.TEST_ENV ?? 'DEV').toUpperCase();

  if (!ALLOWED_ENVS.includes(raw as Env)) {
    throw new Error(`Invalid env "${raw}". Use one of: ${ALLOWED_ENVS.join(', ')}`);
  }
  return raw as Env;
}

export const env: Env = resolveEnv();
